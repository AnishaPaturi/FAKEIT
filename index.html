Experiment 4: Simple E-Commerce Application Using React Router

Aim : Build a simplified e-commerce application using React Router. The application includes Home, Products, Product Details, Contact, Login, Cart, and 404 pages with dynamic routing, protected routes, and navigation links.

Requirements : npm install react-router-dom

Step 1 : In project terminal, install React Router: npm install react-router-dom

Step 2 : Open src/App.jsx and replace with the code below:

src/App.jsx :

// Program 4: Simplified e-commerce app with React Router (works fully offline)
// Needs: npm install react-router-dom   (one-time; afterwards no internet needed)
// Usage: save as src/App.jsx in a Vite/CRA project.
import React, { createContext, useContext, useState } from "react";
import {
  BrowserRouter, Routes, Route, Link, NavLink,
  Navigate, useParams, useNavigate, useLocation,
} from "react-router-dom";

// ---------- Local data (no API) ----------
const PRODUCTS = [
  { id: 1, name: "Laptop", price: 55000, desc: "14-inch, 16GB RAM, 512GB SSD" },
  { id: 2, name: "Headphones", price: 2500, desc: "Wireless, noise cancelling" },
  { id: 3, name: "Smart Watch", price: 4000, desc: "Heart-rate and sleep tracking" },
  { id: 4, name: "Backpack", price: 1200, desc: "Water-resistant, 25L" },
];

// ---------- Global state (auth + cart) ----------
const AppContext = createContext();
const useApp = () => useContext(AppContext);

function AppProvider({ children }) {
  const [user, setUser] = useState(null);
  const [cart, setCart] = useState([]);

  const addToCart = (product) => setCart((c) => [...c, product]);
  const removeFromCart = (index) => setCart((c) => c.filter((_, i) => i !== index));

  return (
    <AppContext.Provider value={{ user, setUser, cart, addToCart, removeFromCart }}>
      {children}
    </AppContext.Provider>
  );
}

// ---------- Protected route ----------
function ProtectedRoute({ children }) {
  const { user } = useApp();
  const location = useLocation();
  if (!user) return <Navigate to="/login" state={{ from: location.pathname }} replace />;
  return children;
}

// ---------- Navigation ----------
function Navbar() {
  const { user, setUser, cart } = useApp();
  const navigate = useNavigate();
  const style = ({ isActive }) => ({
    marginRight: 15, textDecoration: "none",
    fontWeight: isActive ? "bold" : "normal", color: isActive ? "#c0392b" : "#333",
  });

  return (
    <nav style={{ padding: 12, background: "#eee", display: "flex", alignItems: "center" }}>
      <NavLink to="/" style={style}>Home</NavLink>
      <NavLink to="/products" style={style}>Products</NavLink>
      <NavLink to="/contact" style={style}>Contact</NavLink>
      <NavLink to="/cart" style={style}>Cart ({cart.length})</NavLink>
      <span style={{ marginLeft: "auto" }}>
        {user ? (
          <>
            Hi, {user}{" "}
            <button onClick={() => { setUser(null); navigate("/"); }}>Logout</button>
          </>
        ) : (
          <NavLink to="/login" style={style}>Login</NavLink>
        )}
      </span>
    </nav>
  );
}

// ---------- Pages ----------
const Home = () => (
  <div>
    <h2>Welcome to MyShop</h2>
    <p>Browse our products and add your favourites to the cart.</p>
    <Link to="/products">Shop now</Link>
  </div>
);

const Products = () => (
  <div>
    <h2>Products</h2>
    {PRODUCTS.map((p) => (
      <div key={p.id} style={{ border: "1px solid #ccc", padding: 10, margin: "10px 0" }}>
        <h3>{p.name}</h3>
        <p>₹{p.price}</p>
        <Link to={`/products/${p.id}`}>View details</Link>
      </div>
    ))}
  </div>
);

function ProductDetails() {
  const { id } = useParams(); // dynamic routing
  const { addToCart } = useApp();
  const product = PRODUCTS.find((p) => p.id === Number(id));

  if (!product) return <h2>Product not found</h2>;
  return (
    <div>
      <h2>{product.name}</h2>
      <p>{product.desc}</p>
      <p><b>₹{product.price}</b></p>
      <button onClick={() => addToCart(product)}>Add to cart</button>{" "}
      <Link to="/products">Back</Link>
    </div>
  );
}

function Contact() {
  const [sent, setSent] = useState(false);
  return (
    <div>
      <h2>Contact Us</h2>
      {sent ? (
        <p>Thanks! We will get back to you.</p>
      ) : (
        <>
          <input placeholder="Your email" /><br /><br />
          <textarea placeholder="Message" /><br /><br />
          <button onClick={() => setSent(true)}>Send</button>
        </>
      )}
    </div>
  );
}

function Login() {
  const { setUser } = useApp();
  const navigate = useNavigate();
  const location = useLocation();
  const [name, setName] = useState("");
  const [pass, setPass] = useState("");
  const [error, setError] = useState("");

  const handleLogin = () => {
    // Demo credentials: any name with password "1234"
    if (name && pass === "1234") {
      setUser(name);
      navigate(location.state?.from || "/", { replace: true });
    } else {
      setError("Enter a username and the password 1234");
    }
  };

  return (
    <div>
      <h2>Login</h2>
      <input placeholder="Username" value={name} onChange={(e) => setName(e.target.value)} /><br /><br />
      <input type="password" placeholder="Password (1234)" value={pass}
             onChange={(e) => setPass(e.target.value)} /><br /><br />
      <button onClick={handleLogin}>Login</button>
      {error && <p style={{ color: "red" }}>{error}</p>}
    </div>
  );
}

function Cart() {
  const { cart, removeFromCart } = useApp();
  const total = cart.reduce((sum, p) => sum + p.price, 0);
  return (
    <div>
      <h2>Your Cart</h2>
      {cart.length === 0 ? <p>Cart is empty.</p> : (
        <>
          {cart.map((p, i) => (
            <p key={i}>
              {p.name} - ₹{p.price}{" "}
              <button onClick={() => removeFromCart(i)}>Remove</button>
            </p>
          ))}
          <h3>Total: ₹{total}</h3>
        </>
      )}
    </div>
  );
}

const NotFound = () => (
  <div>
    <h2>404 - Page Not Found</h2>
    <Link to="/">Go back home</Link>
  </div>
);

// ---------- App ----------
export default function App() {
  return (
    <AppProvider>
      <BrowserRouter>
        <Navbar />
        <div style={{ padding: 20 }}>
          <Routes>
            <Route path="/" element={<Home />} />
            <Route path="/products" element={<Products />} />
            <Route path="/products/:id" element={<ProductDetails />} />
            <Route path="/contact" element={<Contact />} />
            <Route path="/login" element={<Login />} />
            <Route path="/cart" element={<ProtectedRoute><Cart /></ProtectedRoute>} />
            <Route path="*" element={<NotFound />} />
          </Routes>
        </div>
      </BrowserRouter>
    </AppProvider>
  );
}

Experiment 5: Dynamic React Form Using Configuration Object

Aim : Create a React form where input fields are dynamically generated from a JavaScript configuration object. The form manages user input using state and displays submitted data below the form.

Requirements : Built-in React (No extra packages needed)

Step 1 : Open src/App.jsx and replace with the code below:

src/App.jsx :

// Program 5: Dynamic form generated from a JavaScript configuration object
// No extra packages needed. Usage: save as src/App.jsx
import React, { useState } from "react";

// ---------- Configuration object ----------
const formConfig = {
  title: "Student Registration",
  fields: [
    { name: "fullName", label: "Full Name", type: "text", placeholder: "Enter name", required: true },
    { name: "email", label: "Email", type: "email", placeholder: "Enter email", required: true },
    { name: "age", label: "Age", type: "number", placeholder: "Enter age" },
    { name: "gender", label: "Gender", type: "radio", options: ["Male", "Female", "Other"] },
    { name: "course", label: "Course", type: "select", options: ["BCA", "B.Tech", "MCA", "MBA"] },
    { name: "skills", label: "Skills", type: "checkbox", options: ["React", "Java", "Python"] },
    { name: "about", label: "About You", type: "textarea", placeholder: "Short description" },
  ],
};

// Build initial state from config
const buildInitialState = (config) => {
  const state = {};
  config.fields.forEach((f) => {
    state[f.name] = f.type === "checkbox" ? [] : "";
  });
  return state;
};

export default function App() {
  const [formData, setFormData] = useState(buildInitialState(formConfig));
  const [submitted, setSubmitted] = useState(null);

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    if (type === "checkbox") {
      setFormData((prev) => ({
        ...prev,
        [name]: checked ? [...prev[name], value] : prev[name].filter((v) => v !== value),
      }));
    } else {
      setFormData((prev) => ({ ...prev, [name]: value }));
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    setSubmitted(formData);
  };

  // Render one field depending on its type
  const renderField = (f) => {
    switch (f.type) {
      case "select":
        return (
          <select name={f.name} value={formData[f.name]} onChange={handleChange} required={f.required}>
            <option value="">-- Select --</option>
            {f.options.map((o) => <option key={o} value={o}>{o}</option>)}
          </select>
        );
      case "radio":
        return f.options.map((o) => (
          <label key={o} style={{ marginRight: 12 }}>
            <input type="radio" name={f.name} value={o}
                   checked={formData[f.name] === o} onChange={handleChange} /> {o}
          </label>
        ));
      case "checkbox":
        return f.options.map((o) => (
          <label key={o} style={{ marginRight: 12 }}>
            <input type="checkbox" name={f.name} value={o}
                   checked={formData[f.name].includes(o)} onChange={handleChange} /> {o}
          </label>
        ));
      case "textarea":
        return <textarea name={f.name} placeholder={f.placeholder}
                         value={formData[f.name]} onChange={handleChange} />;
      default:
        return <input type={f.type} name={f.name} placeholder={f.placeholder}
                      value={formData[f.name]} onChange={handleChange} required={f.required} />;
    }
  };

  return (
    <div style={{ maxWidth: 480, margin: "30px auto", fontFamily: "sans-serif" }}>
      <h2>{formConfig.title}</h2>
      <form onSubmit={handleSubmit}>
        {formConfig.fields.map((f) => (
          <div key={f.name} style={{ marginBottom: 14 }}>
            <label style={{ display: "block", fontWeight: "bold", marginBottom: 4 }}>
              {f.label}{f.required && " *"}
            </label>
            {renderField(f)}
          </div>
        ))}
        <button type="submit">Submit</button>
      </form>

      {submitted && (
        <div style={{ marginTop: 25, padding: 12, background: "#f3f3f3" }}>
          <h3>Submitted Data</h3>
          {formConfig.fields.map((f) => (
            <p key={f.name}>
              <b>{f.label}:</b>{" "}
              {Array.isArray(submitted[f.name]) ? submitted[f.name].join(", ") : submitted[f.name]}
            </p>
          ))}
        </div>
      )}
    </div>
  );
}

Experiment 6: React SPA with Routing and API Fetching

Aim : Develop a React single-page application with routing between Home and Users pages. Fetch user data using useEffect and useState, display the data conditionally, and show a loading indicator while data is being fetched.

Requirements : npm install react-router-dom

Step 1 : In project terminal, ensure React Router is installed: npm install react-router-dom

Step 2 : Open src/App.jsx and replace with the code below:

src/App.jsx :

// Program 6: SPA with routing (Home, Users), useEffect + useState,
// conditional rendering and a loading indicator.
// Offline: the "public API" is replaced by a local function that returns
// data after a delay (same behaviour as fetch, but no internet).
// Needs: npm install react-router-dom (one-time). Usage: save as src/App.jsx
import React, { useState, useEffect } from "react";
import { BrowserRouter, Routes, Route, NavLink } from "react-router-dom";

// ---------- Fake API (replaces https://jsonplaceholder.typicode.com/users) ----------
const USERS_DATA = [
  { id: 1, name: "Leanne Graham", email: "leanne@example.com", city: "Hyderabad" },
  { id: 2, name: "Ervin Howell", email: "ervin@example.com", city: "Chennai" },
  { id: 3, name: "Clementine Bauch", email: "clem@example.com", city: "Mumbai" },
  { id: 4, name: "Patricia Lebsack", email: "patricia@example.com", city: "Delhi" },
  { id: 5, name: "Chelsey Dietrich", email: "chelsey@example.com", city: "Pune" },
];

const fetchUsers = (shouldFail = false) =>
  new Promise((resolve, reject) => {
    setTimeout(() => {
      shouldFail ? reject(new Error("Failed to load users")) : resolve(USERS_DATA);
    }, 2000); // simulate 2-second network delay
  });

// ---------- Pages ----------
const Home = () => (
  <div>
    <h2>Home</h2>
    <p>Open the Users page to see data loaded with useEffect.</p>
  </div>
);

function Users() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    fetchUsers()
      .then((data) => setUsers(data))
      .catch((err) => setError(err.message))
      .finally(() => setLoading(false));
  }, []); // runs once when the page mounts

  // Conditional rendering
  if (loading) return <p>Loading users...</p>;
  if (error) return <p style={{ color: "red" }}>{error}</p>;

  return (
    <div>
      <h2>Users</h2>
      <table border="1" cellPadding="8" style={{ borderCollapse: "collapse" }}>
        <thead>
          <tr><th>ID</th><th>Name</th><th>Email</th><th>City</th></tr>
        </thead>
        <tbody>
          {users.map((u) => (
            <tr key={u.id}>
              <td>{u.id}</td><td>{u.name}</td><td>{u.email}</td><td>{u.city}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

// ---------- App ----------
export default function App() {
  const link = ({ isActive }) => ({
    marginRight: 15, fontWeight: isActive ? "bold" : "normal",
  });
  return (
    <BrowserRouter>
      <nav style={{ padding: 12, background: "#eee" }}>
        <NavLink to="/" style={link}>Home</NavLink>
        <NavLink to="/users" style={link}>Users</NavLink>
      </nav>
      <div style={{ padding: 20 }}>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/users" element={<Users />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

Step 3 : Run npm run dev, navigate to "Users" to observe the loading indicator followed by the loaded table.

Note : The code above includes a simulated network delay (2 seconds) that works 100% offline. If you want to fetch directly from jsonplaceholder when internet is available, replace fetchUsers() in useEffect with:

fetch("https://jsonplaceholder.typicode.com/users")
  .then((res) => res.json())
  .then((data) => setUsers(data))
  .catch((err) => setError(err.message))
  .finally(() => setLoading(false));

package com.test.selenium;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;

public class NewTest {
    public static void main(String[] args) {

        try {

            // Connect to H2 Database
            Connection con = DriverManager.getConnection(
                    "jdbc:h2:./StudentDB", "sa", "");

            Statement stmt = con.createStatement();

            // Create table if it does not exist
            stmt.execute("CREATE TABLE IF NOT EXISTS STUDENT ("
                    + "ID INT PRIMARY KEY, "
                    + "NAME VARCHAR(50), "
                    + "EMAIL VARCHAR(100))");

            // Delete previous records
            stmt.execute("DELETE FROM STUDENT");

            // Insert sample record
            stmt.executeUpdate(
                    "INSERT INTO STUDENT VALUES "
                    + "(1,'Ajay','ajay@gmail.com')");

            // Retrieve the record
            ResultSet rs = stmt.executeQuery(
                    "SELECT NAME FROM STUDENT WHERE ID = 1");

            if (rs.next()) {

                String actualName = rs.getString("NAME");
                String expectedName = "Ajay";

                // Custom Check
                if (actualName.equals(expectedName)) {

                    System.out.println("Expected Name : " + expectedName);
                    System.out.println("Actual Name   : " + actualName);
                    System.out.println("CUSTOM CHECK PASSED");

                } else {

                    System.out.println("Expected Name : " + expectedName);
                    System.out.println("Actual Name   : " + actualName);
                    System.out.println("CUSTOM CHECK FAILED");
                }

            } else {

                System.out.println("Record Not Found");
            }

            con.close();

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

package com.test.selenium;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class DynamicDataSubmission {

    public static void main(String[] args) throws Exception {

        WebDriver driver = new ChromeDriver();

        // Open local HTML page
        driver.get("file:///C:/SeleniumFiles/StudentForm.html");

        // Dynamic Test Data
        String[][] data = {
            {"Ajay", "ajay@gmail.com"},
            {"Rahul", "rahul@gmail.com"},
            {"Anita", "anita@gmail.com"}
        };

        for (int i = 0; i < data.length; i++) {

            WebElement name = driver.findElement(By.id("name"));
            WebElement email = driver.findElement(By.id("email"));

            name.clear();
            email.clear();

            name.sendKeys(data[i][0]);
            email.sendKeys(data[i][1]);

            driver.findElement(By.tagName("button")).click();

            String result = driver.findElement(
                    By.id("result")
            ).getText();

            System.out.println("--------------------");
            System.out.println("Test Case : " + (i + 1));
            System.out.println("Name : " + data[i][0]);
            System.out.println("Email : " + data[i][1]);
            System.out.println(result);

            if (result.equals("Registration Successful")) {
                System.out.println("TEST PASS");
            } else {
                System.out.println("TEST FAIL");
            }

            Thread.sleep(1000);
        }

        driver.quit();
    }
}
